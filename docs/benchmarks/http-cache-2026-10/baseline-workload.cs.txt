using System.Net;
using System.Net.Http;
using System.Text;
using BenchmarkDotNet.Attributes;
using Jint.Browser;

namespace Jint.Benchmark;

/// <summary>Measures complete document loads with repeated or distinct cached image URLs.</summary>
/// <remarks>
/// <para>Each document loads 300 deterministic 64 KiB images. Engine construction, parsing and per-consumer
/// copies enter the measurement; sockets and pixel decoding do not.</para>
/// <para>Each row owns its context and warms only its own workload. Control parses sixteen source-less
/// 300-image documents without transport, putting its construction and parsing cost on a similar scale.</para>
/// <para>Warm disk restores persisted entries into bounded memory and includes reopening that context.</para>
/// <para>Request counts, transferred bytes and retained-heap observations are collected outside timing by
/// --validate-http-cache; BenchmarkDotNet measures operation time and allocation separately.</para>
/// </remarks>
[MemoryDiagnoser]
public class BrowserHttpCacheBenchmark
{
    private Browser.Browser _browser = null!;
    private BrowserContext _context = null!;
    private Page _page = null!;
    private Page _control = null!;
    private HttpClient _client = null!;
    private Handler _handler = null!;
    private static readonly string _controlHtml = "<!doctype html><title>control</title>" + string.Concat(Enumerable.Repeat("<img>", 300));
    private string _directory = null!;
    private readonly CacheClock _clock = new();

    [Params(BrowserHttpCacheStorage.Disabled, BrowserHttpCacheStorage.Memory, BrowserHttpCacheStorage.Disk)]
    public BrowserHttpCacheStorage Storage { get; set; }

    [Params(false, true)]
    public bool DistinctUrls { get; set; }

    [GlobalSetup(Targets = [nameof(ColdMiss), nameof(WarmHit), nameof(StaleRevalidation)])]
    public async Task Setup()
    {
        _directory = Path.Combine(Path.GetTempPath(), "jint-cache-benchmark-" + Guid.NewGuid().ToString("N"));
        _handler = new Handler(DistinctUrls, _clock);
        _client = new HttpClient(_handler);
        _browser = new Browser.Browser();
        _context = await _browser.NewContextAsync(Options());
        _page = await _context.NewPageAsync();
        await _page.NavigateAsync("https://cache.test/page");
    }

    [GlobalSetup(Target = nameof(Control))]
    public async Task SetupControl()
    {
        _browser = new Browser.Browser();
        await InitializeControlAsync();
        await Control();
    }

    private async Task InitializeControlAsync()
    {
        _control = await _browser.NewPageAsync();
        await _control.SetContentAsync(_controlHtml);
    }

    private BrowserContextOptions Options()
    {
        var options = new BrowserContextOptions { HttpClient = _client, UrlFilter = uri => uri.Host == "cache.test" };
        return options;
    }

    [Benchmark]
    public async Task ColdMiss()
    {
        // Baseline has no automatic HTTP cache to clear.
        await _page.NavigateAsync("https://cache.test/page");
    }

    [Benchmark]
    public async Task WarmHit()
    {
        if (Storage == BrowserHttpCacheStorage.Disk)
        {
            await _context.DisposeAsync();
            _context = await _browser.NewContextAsync(Options());
            _page = await _context.NewPageAsync();
        }
        await _page.NavigateAsync("https://cache.test/page");
    }

    [Benchmark]
    public async Task StaleRevalidation()
    {
        _clock.Now += TimeSpan.FromHours(2);
        await _page.NavigateAsync("https://cache.test/page");
    }

    [Benchmark]
    public async Task Control()
    {
        for (var i = 0; i < 16; i++) await _control.SetContentAsync(_controlHtml);
    }

    [GlobalCleanup]
    public async Task Cleanup()
    {
        await _browser.DisposeAsync();
        _client?.Dispose();
        if (Directory.Exists(_directory)) Directory.Delete(_directory, true);
    }

    // Counts and heap observations deliberately run separately from BDN's timing and allocation rows.
    internal static async Task<int> ValidateAsync()
    {
        foreach (var storage in Enum.GetValues<BrowserHttpCacheStorage>())
        foreach (var distinct in new[] { false, true })
        {
            var fixture = new BrowserHttpCacheBenchmark { Storage = storage, DistinctUrls = distinct };
            await fixture.Setup();
            await fixture.InitializeControlAsync();
            try
            {
                var warmHeap = GC.GetTotalMemory(forceFullCollection: true);
                // Baseline has no automatic HTTP cache to clear.
                var emptyHeap = GC.GetTotalMemory(forceFullCollection: true);
                Console.WriteLine($"{storage},{distinct},Clear: releasedManagedBytes={warmHeap - emptyHeap} (whole-process GC observation)");
                foreach (var operation in new[] { "ColdMiss", "WarmHit", "StaleRevalidation", "Control" })
                {
                    fixture._handler.Requests = 0;
                    fixture._handler.TransferredBytes = 0;
                    fixture._handler.Validations = 0;
                    var retainedBefore = GC.GetTotalMemory(forceFullCollection: true);
                    var allocatedBefore = GC.GetTotalAllocatedBytes(precise: true);
                    switch (operation)
                    {
                        case "ColdMiss": await fixture.ColdMiss(); break;
                        case "WarmHit": await fixture.WarmHit(); break;
                        case "StaleRevalidation": await fixture.StaleRevalidation(); break;
                        default: await fixture.Control(); break;
                    }
                    var allocated = GC.GetTotalAllocatedBytes(precise: true) - allocatedBefore;
                    var retainedDelta = GC.GetTotalMemory(forceFullCollection: true) - retainedBefore;
                    if (operation != "Control" && !await fixture._page.EvaluateAsync<bool>(
                            "document.images.length === 300 && Array.from(document.images).every(i => i.complete && i.naturalWidth === 1)"))
                        throw new InvalidOperationException("The image workload did not finish correctly.");
                    Console.WriteLine($"{storage},{distinct},{operation}: requests={fixture._handler.Requests}, bodyBytes={fixture._handler.TransferredBytes}, validations={fixture._handler.Validations}, allocatedBytes={allocated}, processRetainedDelta={retainedDelta}");
                    if (operation == "Control" && fixture._handler.Requests != 0)
                        throw new InvalidOperationException("Control reached the transport.");
                    if (storage != BrowserHttpCacheStorage.Disabled && operation == "WarmHit" && fixture._handler.Requests != 1)
                        throw new InvalidOperationException("Warm image bodies were sent over the transport.");
                }
            }
            finally { await fixture.Cleanup(); }
        }
        return 0;
    }

    private sealed class CacheClock : TimeProvider
    {
        internal DateTimeOffset Now = new(2026, 1, 1, 0, 0, 0, TimeSpan.Zero);
        public override DateTimeOffset GetUtcNow() => Now;
    }

    private sealed class Handler : HttpMessageHandler
    {
        internal int Requests;
        internal long TransferredBytes;
        internal int Validations;
        private readonly byte[] _document;
        private readonly byte[] _image = new byte[64 * 1024];
        private readonly CacheClock _clock;
        internal Handler(bool distinct, CacheClock clock)
        {
            _clock = clock;
            var html = new StringBuilder("<!doctype html><title>images</title>");
            for (var i = 0; i < 300; i++) html.Append("<img src='/image/").Append(distinct ? i : 0).Append(".png'>");
            _document = Encoding.UTF8.GetBytes(html.ToString());
            // A valid header is enough for the renderless dimension reader; the remaining bytes are fixed.
            Convert.FromBase64String("iVBORw0KGgoAAAANSUhEUgAAAAEAAAABCAQAAAC1HAwCAAAAC0lEQVR42mP8/x8AAwMCAO+aG1kAAAAASUVORK5CYII=").CopyTo(_image, 0);
        }
        protected override Task<HttpResponseMessage> SendAsync(HttpRequestMessage request, CancellationToken cancellationToken)
        {
            cancellationToken.ThrowIfCancellationRequested();
            var document = request.RequestUri!.AbsolutePath == "/page";
            var validated = !document && request.Headers.IfNoneMatch.Count > 0;
            Interlocked.Increment(ref Requests);
            if (validated) Interlocked.Increment(ref Validations);
            Interlocked.Add(ref TransferredBytes, validated ? 0 : document ? _document.Length : _image.Length);
            var response = new HttpResponseMessage(validated ? HttpStatusCode.NotModified : HttpStatusCode.OK)
            {
                Content = new ByteArrayContent(validated ? [] : document ? _document : _image),
            };
            response.Headers.Date = _clock.Now;
            response.Headers.CacheControl = new() { MaxAge = document ? TimeSpan.Zero : TimeSpan.FromHours(1) };
            response.Headers.ETag = new("\"fixed\"");
            response.Content.Headers.ContentType = new(document ? "text/html" : "image/png");
            return Task.FromResult(response);
        }
    }
}

public enum BrowserHttpCacheStorage { Disabled, Memory, Disk }
