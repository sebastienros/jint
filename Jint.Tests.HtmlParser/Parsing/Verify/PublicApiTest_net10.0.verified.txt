        public static Jint.HtmlParser.Document CreateXml() { }

    {
        public override Jint.HtmlParser.NodeType NodeType { get; }
    }

    {
        Html = 0,
        Xml = 1,
    }

    {
        public string Name { get; }
        public override Jint.HtmlParser.NodeType NodeType { get; }
        public string PublicId { get; }
        public string SystemId { get; }
    }

    {
        public DomException(string name, string message) { }
        public string Name { get; }
    }

    {
        public int AttributeCount { get; }
        public System.Collections.Generic.IEnumerable<Jint.HtmlParser.Attr> Attributes { get; }
        public string LocalName { get; }
        public string? NamespaceUri { get; }

        public void RemoveAttributeNS(string? namespaceUri, string localName) { }
        public Jint.HtmlParser.Attr RemoveAttributeNode(Jint.HtmlParser.Attr attribute) { }
        public void SetAttribute(string name, string value) { }
        public void SetAttributeNS(string? namespaceUri, string qualifiedName, string value) { }
        public Jint.HtmlParser.Attr? SetAttributeNode(Jint.HtmlParser.Attr attribute) { }

    public static class Namespaces
    {
        public const string Html = "http://www.w3.org/1999/xhtml";
        public const string MathMl = "http://www.w3.org/1998/Math/MathML";
        public const string Svg = "http://www.w3.org/2000/svg";
        public const string Xml = "http://www.w3.org/XML/1998/namespace";
        public const string Xmlns = "http://www.w3.org/2000/xmlns/";
    }
    public abstract class Node
    {
        public int ChildCount { get; }
        public System.Collections.Generic.IEnumerable<Jint.HtmlParser.Node> ChildNodes { get; }
        public Jint.HtmlParser.Node? FirstChild { get; }
        public Jint.HtmlParser.Node? LastChild { get; }
        public Jint.HtmlParser.Node? NextSibling { get; }
        public abstract Jint.HtmlParser.NodeType NodeType { get; }
        public Jint.HtmlParser.Document? OwnerDocument { get; }
        public Jint.HtmlParser.Node? ParentNode { get; }
        public Jint.HtmlParser.Node? PreviousSibling { get; }
        public Jint.HtmlParser.Node AppendChild(Jint.HtmlParser.Node child) { }
        public Jint.HtmlParser.Node InsertBefore(Jint.HtmlParser.Node child, Jint.HtmlParser.Node? referenceChild) { }
        public Jint.HtmlParser.Node RemoveChild(Jint.HtmlParser.Node child) { }
        public Jint.HtmlParser.Node ReplaceChild(Jint.HtmlParser.Node child, Jint.HtmlParser.Node oldChild) { }
    }

    {
        Element = 1,

    {
        public ParseLimits() { }
